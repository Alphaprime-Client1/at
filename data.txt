import pandas as pd
from sklearn.neighbors import KNeighborsClassifier
from sklearn.model_selection import train_test_split
from sklearn import metrics

data = pd.read_csv("iris_small.csv", sep=r'\s+', header=None)

X = data.iloc[:, :-1]
y = data.iloc[:, -1]

Xtrain, Xtest, ytrain, ytest = train_test_split(
    X, y, test_size=0.33, random_state=42
)

model = KNeighborsClassifier(n_neighbors=3)

model.fit(Xtrain, ytrain)

ypred = model.predict(Xtest)

for i in range(len(ytest)):
    print("Original:", ytest.iloc[i])
    print("Predicted:", ypred[i])

    if ytest.iloc[i] == ypred[i]:
        print("Correct")
    else:
        print("Wrong")

print("\nConfusion Matrix:")
print(metrics.confusion_matrix(ytest, ypred))

print("\nAccuracy:")
print(metrics.accuracy_score(ytest, ypred))

from pyspark.sql import SparkSession
from pyspark.sql.functions import col, explode, split

spark = SparkSession.builder.appName("Data Analysis").getOrCreate()

# Load CSV Files
twitter = spark.read.csv("twitter.csv", header=True, inferSchema=True)
chat = spark.read.csv("chat.csv", header=True, inferSchema=True)
weblog = spark.read.csv("weblog.csv", header=True, inferSchema=True)

# Twitter Data Analysis
twitter.groupBy("user").count().show()

twitter.withColumn("hashtag", explode(split(col("hashtags"), " "))) \
    .groupBy("hashtag").count().show()

# Chat Data Analysis
chat.groupBy("sender").count().show()

chat.withColumn("word", explode(split(col("message"), " "))) \
    .groupBy("word").count().show()

# Web Log Data Analysis
weblog.groupBy("ip").count().show()
weblog.groupBy("status").count().show()
weblog.groupBy("url").count().show()

# Stop Spark Session
spark.stop()
